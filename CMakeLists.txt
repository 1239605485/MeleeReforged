cmake_minimum_required(VERSION 3.20)
project(MeleeReforged VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MR_BUILD_TESTS "Build host tests" ON)
option(MR_BUILD_MOD "Build KernelLoader plugin" OFF)
add_library(mr_core STATIC src/combat/attack_controller.cpp src/combat/trajectory.cpp src/weapons/weapon_registry.cpp src/weapons/wooden_sword.cpp src/weapons/katana.cpp src/combat/katana_pose.cpp src/collision/blade_sweep.cpp src/render/trail_buffer.cpp src/render/swing_renderer.cpp src/render/wave_renderer.cpp src/carry/registry.cpp src/carry/whip_renderer.cpp)
set_target_properties(mr_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mr_core PUBLIC src)
target_compile_options(mr_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MR_BUILD_TESTS)
 enable_testing()
 add_executable(katana_tests tests/katana_tests.cpp)
 target_link_libraries(katana_tests PRIVATE mr_core)
 add_test(NAME katana COMMAND katana_tests)
 add_executable(carry_system_tests tests/carry_system_tests.cpp)
 target_link_libraries(carry_system_tests PRIVATE mr_core)
 add_test(NAME carry_system COMMAND carry_system_tests)
 add_executable(wooden_features_tests tests/wooden_features_tests.cpp)
 target_link_libraries(wooden_features_tests PRIVATE mr_core)
 add_test(NAME wooden_features COMMAND wooden_features_tests)
 add_executable(export_choreography tools/export_choreography.cpp)
 target_link_libraries(export_choreography PRIVATE mr_core)
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE mr_core)
 add_test(NAME core COMMAND core_tests)
 add_executable(choreography_tests tests/choreography_tests.cpp)
 target_link_libraries(choreography_tests PRIVATE mr_core)
 add_test(NAME choreography COMMAND choreography_tests)
 add_executable(render_bridge_tests tests/render_bridge_tests.cpp)
 target_include_directories(render_bridge_tests PRIVATE src vendor/tefkernel vendor/kernelloader)
 target_compile_definitions(render_bridge_tests PRIVATE __ANDROID__)
 add_test(NAME render_bridge COMMAND render_bridge_tests)
 add_executable(slash_sheet_tests tests/slash_sheet_tests.cpp)
 target_include_directories(slash_sheet_tests PRIVATE src)
 add_test(NAME slash_sheet COMMAND slash_sheet_tests)
 add_executable(interop_tests tests/interop_tests.cpp src/interop/runtime_api.cpp src/interop/tef_imports.cpp)
 target_compile_definitions(interop_tests PRIVATE __ANDROID__)
 target_include_directories(interop_tests PRIVATE src vendor/tefkernel vendor/kernelloader)
 add_library(metadata_fixture SHARED tests/metadata_fixture.cpp)
 set_target_properties(metadata_fixture PROPERTIES OUTPUT_NAME il2cpp)
 target_compile_definitions(interop_tests PRIVATE METADATA_FIXTURE="$<TARGET_FILE:metadata_fixture>")
 add_dependencies(interop_tests metadata_fixture)
 add_test(NAME interop COMMAND interop_tests)
 add_executable(bootstrap_tests tests/bootstrap_tests.cpp src/bootstrap/mod_entry.cpp)
 target_include_directories(bootstrap_tests PRIVATE src vendor/tefkernel vendor/kernelloader)
 add_test(NAME bootstrap COMMAND bootstrap_tests)
endif()
if(MR_BUILD_MOD)
 if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "KernelLoader target requires Android arm64-v8a")
 endif()
 set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
 set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
 set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
 add_subdirectory(vendor/dobby EXCLUDE_FROM_ALL)
 add_library(MeleeReforged SHARED src/bootstrap/mod_entry.cpp src/interop/tef_imports.cpp src/interop/runtime_api.cpp src/interop/game_runtime.cpp src/interop/native_hooks.cpp src/interop/player_adapter.cpp src/interop/melee_hooks.cpp src/interop/draw_adapter.cpp src/interop/effect_atlas.cpp src/interop/holster_adapter.cpp src/interop/sword_wave_adapter.cpp src/interop/wave_draw_adapter.cpp src/interop/item_visuals.cpp src/interop/carry_adapter.cpp src/interop/carry_draw_adapter.cpp)
 target_include_directories(MeleeReforged PRIVATE src vendor/tefkernel vendor/kernelloader)
 target_include_directories(MeleeReforged SYSTEM PRIVATE vendor/dobby/include)
 target_compile_options(MeleeReforged PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
 target_link_libraries(MeleeReforged PRIVATE mr_core dobby_static log dl)
 # Each plugin owns writable API slots whose names also occur in other
 # plugins/kernel exports. Bind internal references locally to avoid ELF
 # symbol interposition between function symbols and pointer variables.
 target_link_options(MeleeReforged PRIVATE -Wl,-Bsymbolic -Wl,--exclude-libs,ALL)
 set_target_properties(MeleeReforged PROPERTIES OUTPUT_NAME "MeleeReforged.android.arm64")
endif()
